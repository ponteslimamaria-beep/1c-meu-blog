const themeToggleBtn = document.getElementById('theme-toggle'); const savedTheme = localStorage.getItem('theme'); if (savedTheme === 'light') { document.body.classList.add('light-mode'); themeToggleBtn.textContent = '🌘 Modo Escuro'; } themeToggleBtn.addEventListener('click', () => { document.body.classList.toggle('light-mode'); if (document.body.classList.contains('light-mode')) { themeToggleBtn.textContent = '🌘 Modo Escuro'; localStorage.setItem('theme', 'light'); } else { themeToggleBtn.textContent = '🌗 Modo de Luz'; localStorage.setItem('theme', 'dark'); } });
